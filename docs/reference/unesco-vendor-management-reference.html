PART 1: SYSTEM OVERVIEW & STATISTICS

1.1 System Purpose & Business Context

System Name: MoUV (Management of UNESCO Vendors)
Domain: https://mouv-qas.hq.int.unesco.org (QAS Environment)
Production Domain: https://mouv.hq.int.unesco.org (assumed)

Primary Purpose:
MoUV serves as a vendor master data management application that acts as an intermediary workflow layer between UNESCO business users and SAP ECC. The system manages the complete lifecycle of vendor master data creation and modification requests.

Core Capabilities:

  1. Vendor Master Data Creation
    • 4 distinct creation flows for different vendor types
    • Structured data entry with field-level validation
    • Support for multiple UNESCO entities (UNES, IIEP, UBO, UIS)
  1. Intelligent Duplicate Detection
    • Fuzzy-matching algorithm using Levenshtein distance
    • Real-time duplicate checking during data entry
    • Similarity threshold: ~70-80% match triggers warning
  1. Country-Based Bank Validation
    • Dynamic field visibility based on country selection
    • Country-specific validation rules (IBAN, SWIFT, routing numbers)
    • Support for SEPA, US ACH, and international payment formats
  1. Workflow Management
    • Multi-stage approval process
    • Role-based routing (Requestor → Vendor Unit → BFM → SAP)
    • Email notifications at each workflow stage
    • SLA tracking and monitoring
  1. Document Management
    • Secure file upload for identification documents
    • Bank certificate storage
    • Virus scanning integration
    • Confidential data handling (marked explicitly in UI)
  1. SAP Integration
    • Direct RFC/BAPI calls to SAP ECC
    • Real-time vendor posting to LFA1, LFBK, LFB1 tables
    • Bidirectional synchronization (status updates from SAP)
    • SAP vendor number retrieval and display
  1. Reporting & Analytics
    • Customizable report generation
    • Drag-and-drop field selection
    • Multiple filter options (country, account group, date range, etc.)
    • Export to Excel/PDF

Business Context:

  • Organization: UNESCO (United Nations Educational, Scientific and Cultural Organization)
  • Primary Users: Finance staff, procurement officers, vendor unit personnel, budget managers
  • Geographic Scope: Global (vendors from 195+ countries)
  • Payment Volume: High-value, international payments requiring robust bank validation
  • Compliance Requirements: UN financial regulations, GDPR, anti-fraud measures

2.3 Authentication Flow

Complete Flow:

1. User → https://mouv-qas.hq.int.unesco.org/
2. Check authentication cookie
3. If not authenticated → Redirect to Azure AD
   https://login.microsoftonline.com/{tenant}/saml2
4. User enters credentials
5. Azure AD validates (Active Directory)
6. MFA prompt (if required by conditional access)
7. Azure AD issues SAML assertion
8. Redirect to MoUV with token
9. MoUV validates token signature
10. Create session, set cookie
11. Display: "Connected as [User Name]"

public async Task<DuplicateCheckResult> SearchExistingVendors(
    VendorSearchRequest request)
{
    // Step 1: Search in Cosmos DB for active vendors
    var activeVendors = await _cosmosClient
        .GetContainer("MoUV", "Vendors")
        .GetItemLinqQueryable<Vendor>()
        .Where(v => v.CompanyCode == request.CompanyCode)
        .ToListAsync();

    // Step 2: Search in SAP via RFC
    var sapVendors = await _sapConnector.SearchVendors(
        request.FamilyName,
        request.GivenName,
        request.DateOfBirth);

    // Step 3: Merge and deduplicate
    var allVendors = activeVendors.Concat(sapVtissueendors).Distinct();

    // Step 4: Calculate similarity scores
    var matches = allVendors
        .Select(v => new {
            Vendor = v,
            Score = CalculateLevenshteinSimilarity(
                request.FamilyName + " " + request.GivenName,
                v.VendorName)
        })
        .Where(m => m.Score >= request.SearchThreshold)
        .OrderByDescending(m => m.Score)
        .Take(20)
        .ToList();

    return new DuplicateCheckResult {
        DuplicatesFound = matches.Any(),
        MatchCount = matches.Count,
        Vendors = matches.Select(m => m.Vendor).ToList()
    };
}

public async Task<FileUploadResult> UploadFile(
    IFormFile file,
    string documentType,
    string requestId)
{
    // Step 1: Validate file
    ValidateFile(file); // Size, type, extension

    // Step 2: Generate unique file ID
    var fileId = $"att-{Guid.NewGuid()}";

    // Step 3: Upload to Azure Blob Storage
    var blobClient = _blobContainerClient.GetBlobClient(fileId);
    await blobClient.UploadAsync(file.OpenReadStream());

    // Step 4: Queue virus scan
    await _serviceBus.SendMessage(new {
        FileId = fileId,
        Action = "VirusScan"
    });

    // Step 5: Store metadata in Cosmos DB
    await _cosmosClient.CreateDocumentAsync(new FileMetadata {
        FileId = fileId,
        FileName = file.FileName,
        DocumentType = documentType,
        RequestId = requestId,
        UploadedBy = _currentUser.DisplayName,
        UploadDate = DateTime.UtcNow,
        Confidential = true
    });

    return new FileUploadResult {
        Success = true,

public async Task<ValidationResult> ValidateRequest(VendorRequest request)
{
    var errors = new List<ValidationError>();

    // 1. Mandatory field checks
    if (string.IsNullOrEmpty(request.PersonalDetails.Profession))
        errors.Add(new ValidationError("Profession is mandatory", "personalDetails.profession"));

    // 2. Bank account presence
    if (!request.BankInformation.Any())
        errors.Add(new ValidationError("At least one bank account required", "bankInformation"));

    // 3. Email format validation
    if (!IsValidEmail(request.Contact.ContactEmail))
        errors.Add(new ValidationError("Invalid email format", "contact.contactEmail"));

    // 4. Bank-specific validations
    foreach (var bank in request.BankInformation)
    {
        // IBAN validation
        if (!string.IsNullOrEmpty(bank.IBAN))
        {
            var ibanValidation = await _bankValidator.ValidateIBAN(bank.IBAN);
            if (!ibanValidation.Valid)
                errors.Add(new ValidationError($"Invalid IBAN: {ibanValidation.Message}",
                    $"bankInformation[{index}].iban"));
        }

        // SWIFT validation
        var swiftValidation = await _bankValidator.ValidateSWIFT(bank.SWIFT);
        if (!swiftValidation.Valid)
            errors.Add(new ValidationError($"Invalid SWIFT: {swiftValidation.Message}",
                $"bankInformation[{index}].swift"));

        // Bank document check
        if (!bank.BankDocuments.Any())
            errors.Add(new ValidationError("Bank documents required",
                $"bankInformation[{index}].bankDocuments"));
    }

    // 5. Duplicate IBAN check
    var duplicateCheck = await CheckDuplicateIBAN(request);
    if (duplicateCheck.DuplicateFound)
        errors.Add(new ValidationError("Vendor with same IBAN already exists",
            "bankInformation.iban"));

    return new ValidationResult
    {
        Valid = !errors.Any(),
        Errors = errors
    };
}

        }

    foreach (var bank in request.BankInformation)

        vendorData.BankData.Add(new BAPIVENDOR_03 {

            BANKL = ExtractBankKey(bank),
            BANKN = bank.AccountNumber,
            IBAN = bank.IBAN,
            SWIFT = bank.SWIFT,
            BKONT = bank.AccountCurrency,
            BVTYP = "1"  // Bank type
        });
    }

    // Step 3: Call RFC via SAP Cloud Connector
    var rfcClient = await _sapConnector.GetRfcClient();
    var function = rfcClient.CreateFunction("BAPI_VENDOR_CREATE");

    function.SetValue("VENDORDATA", vendorData.Header);
    function.SetValue("VENDORCOMPANYDATA", vendorData.CompanyCode);

    var bankTable = function.GetTable("VENDORBANKDATA");
    foreach (var bank in vendorData.BankData)
    {
        bankTable.Append();
        bankTable.SetValue("BANKS", bank.BANKS);
        bankTable.SetValue("BANKL", bank.BANKL);
        bankTable.SetValue("BANKN", bank.BANKN);

    if (!returnTable.Any(r => r.GetValue("TYPE").ToString() == "E"))

            .First(r => r.GetValue("TYPE").ToString() == "E")

public class SAPConnector
{
    private readonly string _destination = "UNE_100";

    public async Task<IRfcFunction> ExecuteBAPI(
        string bapiName,
        Action<IRfcFunction> setupParameters)
    {
        RfcDestination destination = null;

        try
        {
            // Step 1: Get RFC destination
            destination = RfcDestinationManager.GetDestination(_destination);

            // Step 2: Create repository
            var repository = destination.Repository;

            // Step 3: Create function
            var function = repository.CreateFunction(bapiName);

            // Step 4: Set parameters
            setupParameters(function);

            // Step 5: Invoke function
            function.Invoke(destination);

            // Step 6: Check for errors
            var returnTable = function.GetTable("RETURN");
            var errors = returnTable
                .Where(r => r.GetString("TYPE") == "E")
                .ToList();

            if (errors.Any())
            {
                var errorMessage = errors.First().GetString("MESSAGE");
                throw new SAPException($"SAP Error: {errorMessage}");
            }

            return function;
        }
        catch (RfcCommunicationException ex)
        {
            _logger.LogError(ex, "RFC communication error");
            throw new SAPConnectionException("Cannot connect to SAP", ex);
        }
        finally
        {
            destination?.Dispose();
        }
    }
}

// SAP RFC connection pool
public class SAPConnectionPool
{
    private readonly ConcurrentBag<RfcDestination> _pool;
    private readonly int _maxConnections = 10;

    public async Task<RfcDestination> GetConnection()
    {
        if (_pool.TryTake(out var connection))
            return connection;

        if (_pool.Count < _maxConnections)
            return CreateNewConnection();

        // Wait for available connection
        await Task.Delay(100);
        return await GetConnection();
    }

    public void ReturnConnection(RfcDestination connection)
    {
        if (connection.Ping())
            _pool.Add(connection);
        else
            connection.Dispose();
    }
}

public ValidationResult ValidateSWIFT(string swift)
{
    // Remove spaces
    swift = swift.Replace(" ", "").ToUpperInvariant();

    // Length check: 8 or 11 characters
    if (swift.Length != 8 && swift.Length != 11)
        return ValidationResult.Invalid("SWIFT code must be 8 or 11 characters");

    // Format: AAAABBCCDDD
    // AAAA - Bank code (letters)
    // BB - Country code (letters)
    // CC - Location code (letters/digits)
    // DDD - Branch code (letters/digits, optional)

    var pattern = @"^[A-Z]{4}[A-Z]{2}[A-Z0-9]{2}([A-Z0-9]{3})?$";
    if (!Regex.IsMatch(swift, pattern))
        return ValidationResult.Invalid("Invalid SWIFT code format");

    var bankCode = swift.Substring(0, 4);
    var countryCode = swift.Substring(4, 2);
    var locationCode = swift.Substring(6, 2);
    var branchCode = swift.Length == 11 ? swift.Substring(8, 3) : "XXX";

    return ValidationResult.Valid(new {
        BankCode = bankCode,
        CountryCode = countryCode,
        LocationCode = locationCode,
        BranchCode = branchCode
    });
}

public ValidationResult ValidateCBU(string cbu)
{
    // Must be exactly 22 digits
    if (!Regex.IsMatch(cbu, @"^\d{22}$"))
        return ValidationResult.Invalid("CBU must be 22 digits");

    // CBU structure: BBBBSSSC DDDDDDDDDDDDDC
    // BBBB - Bank code (4 digits)
    // SSS - Branch code (3 digits)
    // C - Verification digit 1
    // DDDDDDDDDDDDDD - Account number (14 digits)
    // C - Verification digit 2

    var bankCode = cbu.Substring(0, 4);
    var branchCode = cbu.Substring(4, 3);
    var verif1 = int.Parse(cbu.Substring(7, 1));
    var accountNumber = cbu.Substring(8, 14);
    var verif2 = int.Parse(cbu.Substring(22, 1));

    // Validate verification digit 1
    var calc1 = CalculateVerhoeffDigit(bankCode + branchCode);
    if (calc1 != verif1)
        return ValidationResult.Invalid("CBU verification digit 1 invalid");

    // Validate verification digit 2
    var calc2 = CalculateVerhoeffDigit(accountNumber);
    if (calc2 != verif2)
        return ValidationResult.Invalid("CBU verification digit 2 invalid");

    return ValidationResult.Valid(new {
        BankCode = bankCode,
        BranchCode = branchCode,
        AccountNumber = accountNumber
    });
}

public class RateLimitMiddleware
{
    private readonly IMemoryCache _cache;
    private readonly int _requestsPerMinute = 60;

    public async Task InvokeAsync(HttpContext context)
    {
        var userId = context.User.Identity.Name;
        var cacheKey = $"ratelimit:{userId}";

        var requestCount = _cache.GetOrCreate(cacheKey, entry =>
        {
            entry.AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(1);
            return 0;
        });

        if (requestCount >= _requestsPerMinute)
        {
            context.Response.StatusCode = 429; // Too Many Requests
            await context.Response.WriteAsync("Rate limit exceeded");
            return;
        }

        _cache.Set(cacheKey, requestCount + 1);
        await _next(context);
    }
}

public string MaskIBAN(string iban)
{
    if (string.IsNullOrEmpty(iban) || iban.Length < 8)
        return iban;

    // Show first 4 and last 4, mask middle
    var visible = 8;
    var masked = new string('*', iban.Length - visible);
    return iban.Substring(0, 4) + masked + iban.Substring(iban.Length - 4);
}
